# Pure policy checks are shared by the application and affected-test runner.
set(_recording_tests PlanCursorRecording.cpp TestCursorRecording.cpp)
set(_video_recording_tests PlanVideoRecording.cpp TestVideoRecording.cpp)
if(TARGET Audacity)
   target_sources(Audacity PRIVATE PlanCursorRecording.h PlanCursorRecording.cpp
      CreateCursorCapture.h CreateCursorCapture.cpp)
   target_sources(Audacity PRIVATE PlanVideoRecording.h PlanVideoRecording.cpp)
   if(WIN32 AND audacity_has_video_preview)
      target_sources(Audacity PRIVATE RecordVideoInput.h RecordVideoInput.cpp)
   endif()
   add_unit_test(NAME cursor-recording SOURCES ${_recording_tests})
   add_unit_test(NAME video-recording SOURCES ${_video_recording_tests})
else()
   add_executable(cursor-recording-test ${_recording_tests} ../../tests/Catch2Main.cpp)
   target_compile_features(cursor-recording-test PRIVATE cxx_std_17)
   target_link_libraries(cursor-recording-test PRIVATE Catch2::Catch2)
   catch_discover_tests(cursor-recording-test TEST_PREFIX "cursor-recording::" PROPERTIES TIMEOUT 30)
   add_executable(video-recording-test ${_video_recording_tests} ../../tests/Catch2Main.cpp)
   target_compile_features(video-recording-test PRIVATE cxx_std_17)
   target_link_libraries(video-recording-test PRIVATE Catch2::Catch2)
   catch_discover_tests(video-recording-test TEST_PREFIX "video-recording::" PROPERTIES TIMEOUT 30)
endif()
